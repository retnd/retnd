# Where each gate shard runs (#1497, #1672 P1.3), read by
# scripts/bdtools/release/gate_shards.py. scripts/release/run-gate-shard.sh
# still does the running, and keeps its own refusals (mercury, the ui-browser
# opt-in on the Mac, the 9g fence for selftest-encryption).
#
# Two kinds of line, whitespace-separated. `#` starts a comment.
#
#   host  <name> <ssh target, or - for this machine> [key=value ...]
#   shard <name, or a prefix ending in *> <light|exclusive> <host,host,...> est_min=<n>
#
# A host line's keys (#1672 P1.4, P1.6), each optional:
#   slots=<n>         shards it runs at once (default 1); at most one of them exclusive
#   max_load=<x>      wait before starting a shard while its 1-minute load is over this
#   min_free_gib=<n>  ... or while less than this is free under scratch
#   wait_min=<n>      how long to wait (default 30); past it the shard is INCOMPLETE
#   poll_s=<n>        seconds between looks while waiting (default 30)
#   scratch=<path>    where ci-docker.sh unpacks the tree (default ~, or $TMPDIR here)
# With neither max_load nor min_free_gib the shard starts without a look.
#
# A shard line's hosts are the hosts it may run on, in the order it prefers
# them. A host that is not on its line is never used for it, so a shard that
# should stay off a host leaves it out instead of putting it last. est_min is
# the shard's usual wall time in minutes, from the recorded runs; it only
# orders the work (longest first), it is never a limit.
#
# `exclusive` marks the shards whose suites publish containers on the daemon's
# loopback and load it hard: core/tests/machines, dockerlease and machinegate
# (the core suite, which core-kopia, core-rest, core-docker and structure all
# run a part of), apps/generic/tests/dockercli and the apps/common container
# tests (apps), the e2e gate and the two-machine proof. At most one of them
# runs on a host at a time.
#
# Every shard in gate-shards.txt matches exactly one line here (an exact name
# beats a prefix), every line matches at least one shard, and mercury is
# refused: gate work stays off it.
#
# The command line wins over this file, so losing a host never needs a commit:
#   --host <h>                  every shard on <h> (what the recorder always did)
#   --place <shard>=<h>[,<h>]   one shard's hosts
#   --hosts <h>[,<h>]           only these hosts, for every shard not --place'd
#   --drain <h>                 never this host (repeatable), --place included
# `gate_shards.py place --manifest <file>` prints the placement it would use,
# and `gate_shards.py probe --manifest <file>` asks each host it uses whether
# it answers (ssh, docker), its load and its free space.
#
# A shard that ends INCOMPLETE moves to its next host only when its host then
# fails that probe, which drains the host for the rest of the run, and at most
# twice. A FAILED shard never moves.

# MARS has 28 cpus and the NAS 6. The Mac is shared with everything else I
# run, so it gets one slot and a load ceiling it often sits above.
host mars devshell@192.168.0.100 slots=3 max_load=20 min_free_gib=20 scratch=/work
host nas  rom@192.168.0.10       slots=1 max_load=5  min_free_gib=6  scratch=~/workspace
host mac  -                      slots=1 max_load=20 min_free_gib=20 wait_min=45

# The Docker-heavy shards stay off the Mac: the SFTP connection-count flake
# and the buildx "layer does not exist" fault only show under Docker Desktop.
# --place <shard>=mac still puts one there when MARS and the NAS are gone.
shard core-kopia            exclusive  mars,nas      est_min=7
shard core-rest             exclusive  mars,nas      est_min=18
shard core-docker           exclusive  mars,nas      est_min=2
shard apps                  exclusive  mars,nas      est_min=7
shard e2e                   exclusive  mars,nas      est_min=13
shard two-machine           exclusive  mars,nas      est_min=9
shard structure             exclusive  mars,nas      est_min=13

# js signed on MARS twice; its container runs its own npm ci.
shard js                    light      mars,nas,mac  est_min=30
# The browser specs, and every slice of them once they are split (#1672 P1.1).
# The Mac takes them only with RETND_UI_BROWSER_ALLOW_LOCAL=1, so it is not here.
shard ui-browser*           light      mars,nas      est_min=65

shard static                light      mars,nas,mac  est_min=11
shard selftest-gate         light      mars,nas,mac  est_min=17
shard selftest-compat       light      mars,nas,mac  est_min=6
shard selftest-conformance  light      mars,nas,mac  est_min=3

# The two mutation self-tests need the stall limits that only reach the Mac

# The one shard that needs gh logged in (#1457), which only the Mac is.
