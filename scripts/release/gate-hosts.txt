# Where each gate shard runs (#1497, #1672 P1.3), read by
# scripts/bdtools/release/gate_shards.py. scripts/release/run-gate-shard.sh
# still does the running, and keeps its own refusals (mercury, the ui-browser
# opt-in on the Mac, the 9g fence for selftest-encryption).
#
# Two kinds of line, whitespace-separated. `#` starts a comment.
#
#   host  <name> <ssh target, or - for this machine>
#   shard <name, or a prefix ending in *> <light|exclusive> <host,host,...> est_min=<n>
#
# A shard line's hosts are the hosts it may run on, in the order it prefers
# them. A host that is not on its line is never used for it, so a shard that
# should stay off a host leaves it out instead of putting it last. est_min is
# the shard's usual wall time in minutes, from the recorded runs; it only
# orders the work (longest first), it is never a limit.
#
# `exclusive` marks the shards whose suites publish containers on the daemon's
# loopback and load it hard: core/tests/machines, dockerlease and machinegate
# (the core suite, which core-kopia, core-rest, core-docker and structure all
# run a part of), apps/generic/tests/dockercli and the apps/common container
# tests (apps), the e2e gate and the two-machine proof. At most one of them
# runs on a host at a time.
#
# Every shard in gate-shards.txt matches exactly one line here (an exact name
# beats a prefix), every line matches at least one shard, and mercury is
# refused: gate work stays off it.
#
# The command line wins over this file, so losing a host never needs a commit:
#   --host <h>                  every shard on <h> (what the recorder always did)
#   --place <shard>=<h>[,<h>]   one shard's hosts
#   --hosts <h>[,<h>]           only these hosts, for every shard not --place'd
#   --drain <h>                 never this host (repeatable), --place included
# `gate_shards.py place --manifest <file>` prints the placement it would use.

host mars devshell@192.168.0.100
host nas  rom@192.168.0.10
host mac  -
